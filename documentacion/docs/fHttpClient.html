📚 Descripción

La clase fHttpClient proporciona un cliente HTTP moderno y compatible con PSR-18 para realizar peticiones HTTP de manera eficiente y robusta. Incluye soporte para middleware, autenticación, y una API moderna con acceso tipo propiedades.

🚀 Características Principales:
  • Compatible con estándar PSR-18 para HTTP clients
  • Soporte completo para métodos HTTP (GET, POST, PUT, PATCH, DELETE, HEAD)
  • API moderna con acceso tipo propiedades ($response->status, $response->ok)
  • Sistema de middleware para request/response processing
  • Autenticación integrada (Basic, Bearer, API Key)
  • Manejo avanzado de errores y timeouts
  • Method chaining para configuración fluida

⚡ Sistema de Middleware

Request Middleware

// Middleware para logging
$client->addMiddleware(function($request, $next) {
    error_log("Making request to: " . $request['url']);
    $startTime = microtime(true);
    
    $response = $next($request);
    
    $duration = microtime(true) - $startTime;
    error_log("Request completed in: " . $duration . "s");
    
    return $response;
});

// Middleware para añadir headers automáticamente
$client->addMiddleware(function($request, $next) {
    $request['headers']['X-Request-ID'] = uniqid();
    $request['headers']['X-Timestamp'] = time();
    
    return $next($request);
});

Response Middleware

// Middleware para procesar respuestas
$client->addResponseMiddleware(function($response) {
    // Log error responses
    if (!$response->ok) {
        error_log("HTTP Error {$response->status}: {$response->body}");
    }
    
    // Add custom metadata
    $response->processed_at = time();
    
    return $response;
});

API REST Client

class UserApiClient {
    private $client;
    
    public function __construct($apiKey) {
        $this->client = new fHttpClient('https://api.example.com');
        $this->client->setDefaultHeaders([
            'Authorization' => "Bearer {$apiKey}",
            'Accept' => 'application/json',
            'Content-Type' => 'application/json'
        ]);
    }
    
    public function getUser($id) {
        $response = $this->client->get("/users/{$id}");
        return $response->ok ? $response->json() : null;
    }
    
    public function createUser($userData) {
        $response = $this->client->post('/users', $userData);
        return $response->ok ? $response->json() : null;
    }
    
    public function updateUser($id, $userData) {
        $response = $this->client->patch("/users/{$id}", $userData);
        return $response->ok ? $response->json() : null;
    }
    
    public function deleteUser($id) {
        $response = $this->client->delete("/users/{$id}");
        return $response->ok;
    }
}

Webhook Client

class WebhookClient {
    private $client;
    
    public function __construct() {
        $this->client = new fHttpClient();
        $this->client->setTimeout(10); // Quick timeout for webhooks
        
        // Retry middleware
        $this->client->addMiddleware(function($request, $next) {
            $maxRetries = 3;
            $attempt = 0;
            
            while ($attempt < $maxRetries) {
                try {
                    $response = $next($request);
                    if ($response->ok) {
                        return $response;
                    }
                } catch (Exception $e) {
                    $attempt++;
                    if ($attempt >= $maxRetries) {
                        throw $e;
                    }
                    sleep(pow(2, $attempt)); // Exponential backoff
                }
                $attempt++;
            }
            
            throw new fConnectivityException('Max retries exceeded');
        });
    }
    
    public function sendWebhook($url, $data) {
        try {
            $response = $this->client->post($url, $data, [
                'Content-Type' => 'application/json',
                'X-Webhook-Signature' => $this->generateSignature($data)
            ]);
            
            return $response->ok;
        } catch (Exception $e) {
            error_log("Webhook failed: {$url} - {$e->getMessage()}");
            return false;
        }
    }
    
    private function generateSignature($data) {
        return hash_hmac('sha256', json_encode($data), 'webhook-secret');
    }
}

🔄 Integración con Caché

class CachedApiClient {
    private $client;
    private $cache; // fCache or fRedis instance
    
    public function __construct($cache) {
        $this->client = new fHttpClient('https://api.example.com');
        $this->cache = $cache;
        
        // Response caching middleware
        $this->client->addResponseMiddleware(function($response) {
            if ($response->ok && $response->getHeader('Cache-Control')) {
                $cacheKey = 'http_' . md5($response->getRequestUrl());
                $ttl = $this->parseCacheControl($response->getHeader('Cache-Control'));
                $this->cache->set($cacheKey, $response->body, $ttl);
            }
            return $response;
        });
    }
    
    public function getWithCache($endpoint, $ttl = 3600) {
        $cacheKey = 'http_' . md5($endpoint);
        
        // Try cache first
        $cached = $this->cache->get($cacheKey);
        if ($cached !== null) {
            return json_decode($cached, true);
        }
        
        // Fetch from API
        $response = $this->client->get($endpoint);
        if ($response->ok) {
            $data = $response->json();
            $this->cache->set($cacheKey, json_encode($data), $ttl);
            return $data;
        }
        
        return null;
    }
}
